from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import (
    ApplicationBuilder,
    CommandHandler,
    CallbackQueryHandler,
    ContextTypes,
)

TOKEN = "8858509512:AAE8e3qib8Jp71CclaPJR9EYC7bvD3nyUcI"

plans = {
    "1 گیگ": "127 هزار تومان",
    "2 گیگ": "254 هزار تومان",
    "5 گیگ": "639 هزار تومان",
    "10 گیگ": "1 میلیون و 270 هزار تومان",
    "20 گیگ": "2 میلیون و 500 هزار تومان",
}

ADMIN = "@FHSLOP"

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keyboard = [
        [InlineKeyboardButton("1 گیگ", callback_data="1 گیگ")],
        [InlineKeyboardButton("2 گیگ", callback_data="2 گیگ")],
        [InlineKeyboardButton("5 گیگ", callback_data="5 گیگ")],
        [InlineKeyboardButton("10 گیگ", callback_data="10 گیگ")],
        [InlineKeyboardButton("20 گیگ", callback_data="20 گیگ")],
    ]

    text = """
👋 به ربات فروش VPN خوش آمدید

📦 پلن مورد نظر خود را انتخاب کنید:
"""

    reply_markup = InlineKeyboardMarkup(keyboard)

    await update.message.reply_text(
        text,
        reply_markup=reply_markup
    )

async def plan_selected(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()

    plan = query.data
    price = plans[plan]

    text = f"""
✅ پلن انتخابی: {plan}

💰 قیمت: {price}

📩 برای دریافت اطلاعات پرداخت و خرید:
{ADMIN}

بعد از پرداخت، رسید را برای ادمین ارسال کنید.
"""

    await query.message.reply_text(text)

app = ApplicationBuilder().token(TOKEN).build()

app.add_handler(CommandHandler("start", start))
app.add_handler(CallbackQueryHandler(plan_selected))

print("Bot is running...")
app.run_polling()
